### 1. 身份认证系统 (Authentication)

#### 登录流程（Login.vue）
```vue
<!-- 用户账户和密码输入表单 -->
<template>
    <div class="login-container">
        <div class="login-panel">
            <!-- 左侧图片区域 -->
            <div style="width: 300px;padding: 40px;">
                <img src="/bag.png" style="width: 350px;" />
            </div>
            <!-- 右侧登录表单区域 -->
            <div class="right-login">
                <h2>校园二手交易平台</h2>
                <!-- 账号输入框 -->
                <input v-model="act" class="act" placeholder="输入账号" />
                <!-- 密码输入框 -->
                <input v-model="pwd" class="pwd" type="password" placeholder="输入密码" />
                <!-- 登录按钮 -->
                <span class="login-btn" @click="login">立即登录</span>
                <!-- 跳转到注册页面 -->
                <span class="no-act" @click="toDoRegister">点此注册</span>
            </div>
        </div>
    </div>
</template>

<script>
import request from "@/utils/request.js";
import { setToken } from "@/utils/storage.js";
import md5 from 'js-md5';

export default {
    name: "Login",
    data() {
        return {
            act: '',        // 账户
            pwd: '',        // 密码
        }
    },
    methods: {
        async login() {
            // 1. 校验账户和密码不能为空
            if (!this.act || !this.pwd) {
                this.$swal.fire({
                    title: '填写校验',
                    text: '账号或密码不能为空',
                    icon: 'error',
                    showConfirmButton: false,
                    timer: 1300,
                });
                return;
            }
            
            // 2. 密码使用MD5加密（双重加密）
            const hashedPwd = md5(md5(this.pwd));
            const paramDTO = { userAccount: this.act, userPwd: hashedPwd };
            
            try {
                // 3. 发送登录请求到后端
                const { data } = await request.post(`user/login`, paramDTO);
                
                // 4. 检查响应状态
                if (data.code !== 200) {
                    this.$swal.fire({
                        title: '登录失败',
                        text: data.msg,
                        icon: 'error',
                        showConfirmButton: false,
                        timer: 1300,
                    });
                    return;
                }
                
                // 5. 存储token到sessionStorage
                setToken(data.data.token);
                
                // 6. 根据用户角色进行页面跳转
                setTimeout(() => {
                    const { role } = data.data;
                    this.navigateToRole(role);  // role: 1=管理员, 2=普通用户
                }, 1300);
                
            } catch (error) {
                console.error('登录请求错误:', error);
                this.$message.error('登录请求出错，请重试！');
            }
        },
        
        // 角色导航
        navigateToRole(role) {
            switch (role) {
                case 1:
                    this.$router.push('/admin');  // 管理员后台
                    break;
                case 2:
                    this.$router.push('/user');   // 普通用户前台
                    break;
                default:
                    console.warn('未知的角色类型:', role);
                    break;
            }
        }
    }
};
</script>
```

#### 令牌存储（storage.js）
```javascript
/**
 * localStorage和sessionStorage管理
 * 用于存储用户会话信息、搜索记录等
 */

const TOKEN_KEY = "token"
const USER_INFO = "userInfo";
const SEARCH_KEY = "searchKey";
const PRODUCT_INFO = "productInfo";

/**
 * 获取JWT令牌
 * @returns {string} token值
 */
export function getToken() {
    return sessionStorage.getItem(TOKEN_KEY);
}

/**
 * 设置JWT令牌
 * @param {string} token 令牌值
 */
export function setToken(token) {
    sessionStorage.setItem(TOKEN_KEY, token);
}

/**
 * 获取用户信息
 * @returns {object} 用户对象
 */
export function getUserInfo() {
    const jsonUserInfo = sessionStorage.getItem(USER_INFO);
    return JSON.parse(jsonUserInfo);
}

/**
 * 获取商品查询缓存
 * @returns {object} 商品信息
 */
export function getProductInfo() {
    const jsonProduct = sessionStorage.getItem(PRODUCT_INFO);
    return JSON.parse(jsonProduct);
}

/**
 * 获取搜索关键词缓存
 * @returns {string} 搜索关键词
 */
export function getSearchKey() {
    return sessionStorage.getItem(SEARCH_KEY);
}
```

```javascript
        const token = getToken();
        
        // 定义不需要登录就能访问的路由
        const releasePaths = ['/main', '/product'];

                next("/login");

});
```

### 3. HTTP请求拦截（request.js）

#### Axios全局配置与拦截器
```javascript
import axios from "axios"
import { getToken } from "@/utils/storage.js";

// API基础地址
const URL_API = 'http://localhost:21090/api/campus-product-sys/v1.0'

/**
 * 创建Axios实例
 * 配置基础URL和请求超时时间
 */
const request = axios.create({
    baseURL: URL_API,
    timeout: 8000  // 8秒请求超时
});

/**
 * 全局请求拦截器
 * 在每个HTTP请求头中添加JWT令牌
 */
request.interceptors.request.use(config => {
    const token = getToken();
    
    // 如果存在token，将其添加到请求头中
    if (token !== null) {
        config.headers["token"] = token;
    }
    
    return config;
}, error => {
    return Promise.reject(error);
});

export default request;
```

```vue
import Editor from "@/components/Editor"

export default {
    components: { Editor },
    name: 'PostProduct',

        return {
            oldLevel: 9,                 // 默认九成新
            inventory: 1,                // 默认库存1件
            product: {},                 // 商品数据对象

            dialogVisible: false,        // 图片预览对话框显示状态

    methods: {
        /**
         * 发布商品
         * 1. 验证表单必填项
         * 2. 组装商品数据
         * 3. 调用后端API保存商品
         */
        postProduct() {
            // 商品名校验
            if (this.product.name === undefined) {
                this.$notify.info({
                    duration: 1000,
                    title: '填写提醒',
                    message: '商品标题不能为空'
                });
                return;
            }
            
            // 价格校验
            if (this.product.price === undefined) {
                this.$notify.info({
                    duration: 1000,
                    title: '填写提醒',
                    message: '商品价格不能为空'
                });
                return;
            }
            
            // 封面图校验
            if (this.coverList.length === 0) {
                this.$notify.info({
                    duration: 1000,
                    title: '上传提醒',
                    message: '请上传封面'
                });
                return;
            }
            
            // 将图片数组转为逗号分隔的字符串
            this.product.coverList = this.coverList.join(',');
            
            // 发送POST请求到后端保存商品
            this.$axios.post('/product/save', this.product)
                .then(res => {
                    const { data } = res;
                    if (data.code === 200) {
                        this.$notify.success({
                            title: '发布操作',
                            message: data.msg

                        // 清空表单，准备发布下一个商品

                    }

                        title: '发布操作',
                        message: error
                    });

                });

        onReceive(detail) {
```

## 三、核心技术实现特点

### 1. **双密钥加密方案**
```javascript
// MD5双重加密：先对原始密码加密，再次加密结果
const hashedPwd = md5(md5(this.pwd));
```
提高密码安全性，确保用户信息保护。

### 2. **Token认证机制**
- 使用JWT令牌进行用户身份认证
- Token存储在sessionStorage中
- 请求拦截器自动添加token到HTTP头部

### 3. **路由守卫与权限控制**
```javascript
router.beforeEach((to, from, next) => {
    if (to.meta.requireAuth) {
        const token = getToken();
        if (token !== null) {
            next();
        } else {
            next("/login");
        }
    }
});
```
实现前端级别的权限控制，防止未授权访问。

### 4. **组件化架构**
- 富文本编辑器的独立封装（Editor.vue）
- 图表组件化（LineChart.vue）
- 可复用的UI组件

### 5. **响应式设计**
- 使用Element UI栅格系统
- 适配不同分辨率的展示效果
- `:span="6"` 网格布局实现自适应

### 6. **异步数据加载**
- Axios异步HTTP请求
- Promise和async/await处理
- 错误捕获和用户友好的错误提示

### 7. **状态管理**
- sessionStorage存储会话信息
- 组件data管理本地状态
- 通过props和事件进行组件通信
